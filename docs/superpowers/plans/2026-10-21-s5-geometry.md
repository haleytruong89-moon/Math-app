# S5 — Geometry proofs Implementation Plan

> **For agentic workers:** Implement this plan task-by-task.
> Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Let the student learn to *find* and *write* a geometry proof in the app. This adds three auto-graded item types — backward analysis, the C3 fill-in proof and the C2 order-the-steps — plus a reviewed "Sổ tay suy ngược" handbook and a "draw it yourself first" figure gate.

**Architecture:**
- The three types are new members of the content `Item` union (zod, `src/lib/content/schema.ts`) and are graded by the pure `gradeItem` (`grade.ts`).
- Lines and choices reach the browser in a deterministic shuffled order (`shuffle.ts`), so no answer leaves the server. The server maps the shown positions back.
- New client components live in `src/components/lesson/geometry/`. They plug into `ItemInput` and `PracticeStage`, which is how practice, exit, review, homework and "Làm thử" pick them up.
- No SQL: answers stay short strings in `attempts.answer` (≤ 200 chars).

**Tech Stack:**
- Next.js 16 App Router (read `node_modules/next/dist/docs/` before using a Next API).
- React 19, TypeScript, zod, Tailwind v4 tokens.
- `node --test` for unit tests.
- Playwright e2e on the fixture registry (`e2e/run.sh`).

**Spec:** `docs/superpowers/specs/2026-10-21-s5-geometry-design.md` (+ `docs/SPEC.md` §0 rows of 21/10). Mockups: design canvas "Bàn làm việc gia sư", row "S5 · Học hình".

```ts
// S5: the order in which C2 lines and C3 choices are shown. Deterministic (seeded by the item id), so the server
// can map a shown position back to the authored one; never the identity, so the authored (right) order never shows.
// Pure and dependency-free: it is also bundled for the browser through grade.ts.

/** FNV-1a, 32 bit. */
function hash(s: string): number {
  let h = 0x811c9dc5;
  for (let i = 0; i < s.length; i++) {
    h ^= s.charCodeAt(i);
    h = Math.imul(h, 0x01000193) >>> 0;
  }
  return h >>> 0;
}

/** mulberry32 */
function rng(seed: number): () => number {
  return () => {
    seed = (seed + 0x6d2b79f5) >>> 0;
    let t = seed;
    t = Math.imul(t ^ (t >>> 15), t | 1);
    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

/** `order[p]` = the authored index shown at position p. */
export function displayOrder(seed: string, n: number): number[] {
  const order = Array.from({ length: n }, (_, i) => i);
  const r = rng(hash(seed));
  for (let i = n - 1; i > 0; i--) {
    const j = Math.floor(r() * (i + 1));
    [order[i], order[j]] = [order[j], order[i]];
  }
  if (n >= 2 && order.every((v, i) => v === i)) order.push(order.shift()!);
  return order;
}
```

export function afterOf(lines: { after?: number[] }[], i: number): number[] {
  return lines[i].after ?? (i === 0 ? [] : [i - 1]);

  understood: boolean;
  correct: boolean;
  errorCode?: string;
  feedback?: string;

  steps?: boolean[];

const NOT: GradeResult = { understood: false, correct: false };

export function gradeItem(item: GradableItem, raw: string, opts: { blanksWrong?: boolean } = {}): GradeResult {
  if (item.type === "steps" || item.type === "analysis") {
    const parts = raw.split(STEP_SEPARATOR);
    if (parts.length !== item.steps.length) return NOT;
    const each = item.steps.map((st, i) => gradeSimple(st.item, parts[i]));
    if (!opts.blanksWrong && each.some((g) => !g.understood)) return NOT;
    const ok = each.map((g) => g.understood && g.correct);
    const first = each.find((g) => !(g.understood && g.correct));
    if (!first) return { understood: true, correct: true, steps: ok };
    return { understood: true, correct: false, errorCode: first.errorCode, feedback: first.feedback, review: first.review, steps: ok };
  }
  if (item.type === "proof") return gradeProof(item, raw, opts);
  if (item.type === "order") return gradeOrder(item, raw);
  return gradeSimple(item, raw);
}

function gradeProof(item: ProofGradable, raw: string, opts: { blanksWrong?: boolean }): GradeResult {
  const order = displayOrder(item.id, item.bank.length);
  const parts = raw.split(STEP_SEPARATOR);
  if (parts.length !== item.blanks.length) return NOT;
  const picked = parts.map((p) => (/^\d$/.test(p.trim()) && Number(p) < order.length ? order[Number(p)] : null));
  if (picked.every((x) => x === null) || (!opts.blanksWrong && picked.some((x) => x === null))) return NOT;
  const ok = picked.map((x, i) => x === item.blanks[i].correct);
  const first = ok.indexOf(false);
  if (first < 0) return { understood: true, correct: true, parts: ok };
  const k = picked[first];
  const m = k === null ? undefined : item.blanks[first].mistakes?.[String(k)];
  return { understood: true, correct: false, parts: ok, errorCode: m?.code, feedback: m?.feedback, review: m?.review };
}

function gradeOrder(item: OrderGradable, raw: string): GradeResult {
  const n = item.lines.length;
  const total = n + item.distractors.length;
  const s = raw.trim();
  if (!s) return NOT;
  const pos = s.split(ORDER_SEPARATOR).map((p) => (/^\d{1,2}$/.test(p.trim()) ? Number(p) : -1));
  if (pos.some((p) => p < 0 || p >= total) || new Set(pos).size !== pos.length) return NOT;
  const order = displayOrder(item.id, total);
  const src = pos.map((p) => order[p]); // < n: a line; ≥ n: distractor (src − n)
  const placedAt = new Map(src.map((v, i) => [v, i]));
  for (let i = 0; i < src.length; i++) {
    const v = src[i];
    if (v >= n) {
      const d = item.distractors[v - n];
      return { understood: true, correct: false, at: i, errorCode: d.code, feedback: d.feedback };
    }
    const missing = afterOf(item.lines, v).find((a) => !(placedAt.has(a) && placedAt.get(a)! < i));
    if (missing !== undefined)
      return { understood: true, correct: false, at: i, errorCode: "E09", feedback: `Dòng này cần «${item.lines[missing].text}» đứng trước.` };
  }
  if (src.length < n) return { understood: true, correct: false, at: src.length, errorCode: "E09", feedback: "Lời giải còn thiếu bước — chưa đi tới kết luận." };
  return { understood: true, correct: true };
}

```ts
// S5: "Sổ tay suy ngược" — goal → ways → signs in the problem, plus recurring figures with their ready-made facts.
// Reviewed and approved like content (draft until the parent approves in a PR). Pure for node tests.
import { z } from "zod";
import { SvgFigure } from "./schema.ts";

const Id = z.string().regex(/^[a-z0-9-]+$/);
const Text = z.string().trim().min(1);
export const SoTay = z
  .object({
    status: z.enum(["draft", "approved"]),
    approvedOn: z.string().regex(/^\d{4}-\d{2}-\d{2}$/).optional(),
    goals: z.array(z.object({ id: Id, title: Text, tip: Text, ways: z.array(z.object({ how: Text, sign: Text, units: z.array(Id).min(1) }).strict()).min(1) }).strict()).min(1),
    figures: z.array(z.object({ id: Id, title: Text, figure: SvgFigure.optional(), facts: z.array(Text).min(1) }).strict()),
  })
  .strict()
  .refine((s) => s.status !== "approved" || !!s.approvedOn, { message: "approved cần approvedOn" });
export type SoTay = z.infer<typeof SoTay>;

export function checkSoTay(s: SoTay, unitIds: Set<string>): string[] {
  const out: string[] = [];
  for (const g of s.goals) for (const w of g.ways) for (const u of w.units) if (!unitIds.has(u)) out.push(`so-tay ${g.id}: bài "${u}" không có trong lộ trình`);
  const ids = [...s.goals.map((g) => g.id), ...s.figures.map((f) => f.id)];
  if (new Set(ids).size !== ids.length) out.push("so-tay: id trùng");
  return out;
}
```
